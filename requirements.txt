# Float runs on nothing but the Python standard library.
# Everything below is optional — install only what you want.
#
# Install everything at once:
#   pip install -r requirements.txt

# Encrypts saved API keys with AES-256-GCM instead of the stdlib fallback.
# Strongly recommended.
cryptography>=42.0

# Lets Float move the mouse and type for you (the "use the computer" tool).
# Not needed if you never plan to use that feature.
pyautogui>=0.9.54

# Lets pressing Escape actually interrupt a running "use the computer" task.
# Without it, the stop button and API still work - only the Escape shortcut
# is unavailable. On Linux this package may need to run with elevated
# privileges to watch the keyboard globally.
keyboard>=0.13.5

# Lets Float type non-English text (Hindi and other Indic scripts, accents,
# emoji) correctly during "use the computer" tasks, via clipboard paste.
# Without it, only characters pyautogui recognises as keys will be typed.
pyperclip>=1.8.2
